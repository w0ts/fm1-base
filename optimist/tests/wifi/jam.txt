# MODE JAM in the trial (device/wifi_own.c wc_jam_pick; host simulator, OPT_HOST_BT=1 OPT_HOST_WIFI=1 with profiles/dev-wifi.config:
# optimist/run.sh's wifi run). The simulated network is an Optimist access point (FM1-AB12) with its default password
# (wifi_ap_default_key); a Link peer on it. MODE AUTO hears it and ignores it; MODE JAM joins it by itself with the derived
# key, Link runs although SYNC > LINK is OFF (the stored value untouched) and goes back to it when JAM is left; an access
# point that has a typed password refuses the derived key: one try, JAM says so (wifi_jam 2) and leaves it alone
wait 1500
master 2048
wait 100
wifi_net FM1-AB12 default 6
link_peer 120
linkset 0
wifi mode AUTO
wait 4000
wifi scan
wait 4000
expect wifi_mode 1
expect wifi_found 1
expect wifi_step 2
expect wifi_jam 0
expect link_peers 0
# JAM: joined by itself, nothing typed, nothing saved by it
wifi mode JAM
wait 12000
expect wifi_mode 7
expect wifi_step 8
expect wifi_ip_last 2
expect wifi_jam 1
expect link_stored 0
expect link_eff 1
expect link_peers 1
# leaving JAM: the LINK setting is back (OFF), the stored value was never touched
wifi mode AUTO
wait 500
expect link_stored 0
expect link_eff 0
# an access point whose owner typed a password: the derived key is refused
wifi leave
wifi_net FM1-AB12 typedkey1 6
wait 500
wifi mode JAM
wait 25000
expect wifi_mode 7
expect wifi_step 9
expect wifi_jam 2
expect link_eff 1
wifi mode AUTO
wifi_net off
